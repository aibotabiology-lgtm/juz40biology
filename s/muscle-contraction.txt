function drawFil(o){// o:{L,s,thick:[ys],thin:[ys],hl,tr,tk,clipH,heads,t,wig}
 const s=o.s,L=o.L*s,hT=1.6*s/2,aL=1.0*s;

  const hz=Math.max(0,L-2*aL)/2;if(hz>0){ctx.fillStyle=o.bgH;ctx.fillRect(m-hz,-o.clipH,2*hz,2*o.clipH)}}
