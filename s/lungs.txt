  const CARD = () => S.W >= 640 ? { side: 'right', w: 232, h: 128 } : { side: 'bottom', w: S.W - 20, h: 84 };
  function stepPad(k) {
    const st = STEPS[k], pad = { l: 14, r: 14, t: 14 + (st.tag ? 32 : 0), b: 14 };
    if (st.card) { const c = CARD(); if (c.side === 'right') pad.r += c.w + 10; else pad.b += c.h + 8; }
    if (st.scene === 'body' && k <= 1 && S.W >= 640) { pad.l += 40; pad.r += 40; }
    return pad;
  }
  function fitBox(b, pad) {
    const aw = Math.max(40, S.W - pad.l - pad.r), ah = Math.max(40, S.H - pad.t - pad.b);
    const z = Math.min(aw / (b.x1 - b.x0), ah / (b.y1 - b.y0));
    return { x: (b.x0 + b.x1) / 2 - (pad.l + aw / 2 - S.W / 2) / z, y: (b.y0 + b.y1) / 2 - (pad.t + ah / 2 - S.H / 2) / z, z };
  }
  const stepCam = k => fitBox(STEPS[k].box(), stepPad(k));
  function camLerp(a, b, p) {
    const z = Math.exp(lerp(Math.log(a.z), Math.log(b.z), p)), w = Math.abs(Math.log(b.z / a.z)) < 1e-3 ? p : (1 / a.z - 1 / z) / (1 / a.z - 1 / b.z);
    return { x: lerp(a.x, b.x, w), y: lerp(a.y, b.y, w), z };
  }

  function iaInit(k, rest) {
    const st = STEPS[k], kind = st.ia;
    S.ia = kind ? { kind, st: rest && kind !== 'tip' ? 'done' : 'wait', v: rest ? 1 : 0, t0: S.t, reach: rest ? ORG.length - 1 : -1, tint: 0, mol: -1 } : null;
    if (st.scene === 'gas') gasReset(k, rest);
    if (kind === 'o2') S.ia.mol = 0;
    if (kind === 'co2') S.ia.mol = 14;
    if (kind === 'air') { S.ia.s = rest ? 1 : 0; S.ia.reach = rest ? ORG.length - 1 : 0; S.ia.tint = rest ? 1 : 0; }
    if (st.scene === 'spiro') {
      SPR.u = k === 11 ? (rest ? 3 : 0) : k === 12 ? (rest ? 3.8 : 3) : (rest ? 7.6 : 3.8);
      SPR.a1 = k >= 12 || rest ? 1 : 0; SPR.a2 = k === 13 || (k === 12 && rest) ? 1 : 0; SPR.a4 = SPR.a8 = k === 13 && rest ? 1 : 0;
      SPR.dv = 0; SPR.auto = null;
    }

  function iaAuto() {   // главная кнопка вместо руки
    const I = S.ia; if (!I || I.st === 'done' || I.st === 'auto') return;

    if (I.kind === 'o2' || I.kind === 'co2') { const M = G.mols[I.mol]; M.st = 'home'; if (I.kind === 'o2') o2Bound(false); else co2Out(false); }
    if (I.kind === 'up' || I.kind === 'down') I.u0 = SPR.u;

  }
  function iaDone() { const I = S.ia; I.st = 'done'; I.v = 1; I.tDone = S.t; S.labKey = null; }
  function iaStep(dt) {
    const I = S.ia, k = S.step;
    if (I) {
      if (I.kind === 'air') {
        if (I.st === 'auto') { I.s = Math.min(1, I.s0 + (S.t - I.t0) * 0.32); if (I.s >= 1) iaDone(); }
        const r = orgAt(I.s); if (r > I.reach) { I.reach = r; S.labKey = null; }
        if (I.st === 'done') I.tint = Math.min(1, I.tint + dt * 1.2);
      }
      if (I.kind === 'dia' || I.kind === 'rib') {
        if (I.st === 'auto') { I.v = lerp(I.v0, 1, smooth((S.t - I.t0) / 1.4)); if (S.t - I.t0 >= 1.4) iaDone(); }
        else if (I.st === 'wait' && I.v > 0) I.v = Math.max(0, I.v - dt * 1.4);   // отпустил на полпути — пружинит обратно
      }
      if (I.kind === 'o2') {
        if (I.st === 'auto' || I.st === 'bound') { const M = G.mols[0]; if (M.st === 'in') { G.load[4] = Math.min(1, G.load[4] + dt / 1.3); if (G.load[4] >= 1 && G.mols[6].st === 'in') iaDone(); } }
      }
      if (I.kind === 'co2') {
        if (I.st === 'auto' || I.st === 'bound') { if (G.mols[14].st === 'home' && G.mols[13].st === 'home' && G.mols[15].st === 'home' && S.t - I.t0 > 3.1) { iaDone(); } }
        if (I.st === 'done') { G.flowV = Math.min(34, G.flowV + dt * 20); G.exh = Math.min(1, G.exh + dt); }
      }
      if (I.kind === 'up' || I.kind === 'down') {
        if (I.st === 'auto') { const p = smooth((S.t - I.t0) / 1.6), a = I.kind === 'up' ? 3.8 : 5.0; SPR.u = lerp(I.u0, a, p); if (p >= 1) { iaDone(); } }
        if (I.st === 'done' && I.kind === 'up') SPR.a2 = Math.min(1, SPR.a2 + dt * 2.5);
        if (I.st === 'done' && I.kind === 'down') { SPR.a4 = Math.min(1, SPR.a4 + dt * 2.5); if (SPR.u < 7.6) SPR.u = Math.min(7.6, SPR.u + dt * 0.62); if (SPR.u > 5.6) SPR.a8 = Math.min(1, SPR.a8 + dt * 2); }
      }
    }
    // механика дыхания на уровне тела
    if (STEPS[k].scene === 'body') {
      let d = 0, r = 0;
      const tt = S.t - S.stepT0 - (S.fly ? 99 : 0);
      if (k <= 1) { const b = (1 - Math.cos(TAU * S.clk / 4.6)) / 2; d = 0.28 * b; r = 0.18 * b; }
      if (k === 3) { d = I.v; }
      if (k === 4) { d = 1; r = I.v; }
      if (k === 5) {
        const e = S.ex0 === undefined ? 0 : S.t - S.ex0;
        if (e < 2.6) { const p = smooth(e / 2.6); d = (1 - p) * S.exD; r = (1 - p) * S.exR; }
        else { const b = (1 - Math.cos(TAU * (e - 2.6) / 4.6)) / 2; d = 0.55 * b; r = 0.4 * b; }
        void tt;
      }
      S.m.d = d; S.m.r = r;
    }
    if (STEPS[k].scene === 'spiro' && k === 11 && S.ia === null) {
      if (!S.fly && !S.tr && SPR.u < 3) { SPR.u = Math.min(3, SPR.u + dt / 1.7); if (SPR.u >= 3) S.labKey = null; }
      if (SPR.u >= 3) SPR.a1 = Math.min(1, (SPR.a1 || 0) + dt * 2.5);
    }
    // газообмен: кровоток, насыщение эритроцитов
    if (STEPS[k].scene === 'gas') {
      G.flowX += G.flowV * dt;
      for (let i = 0; i < RBCN; i++) { const x = rbcX(i); if (x < G.prevX[i] - 100) G.load[i] = 0; if (G.flowV > 0.5) G.load[i] = Math.max(G.load[i], smooth((x + 90) / 150)); G.prevX[i] = x; }
    }
  }
  // объём грудной клетки → поток воздуха
  function flowStep(dt) {
    if (STEPS[S.step].scene === 'body') {
      const v = 0.6 * S.m.d + 0.4 * S.m.r, f = dt > 0 ? (v - S.vol) / dt : 0; S.vol = v;
      S.flow += (clamp(f, -2, 2) - S.flow) * Math.min(1, dt * 6);
      S.airPh += S.flow * dt * 0.55;
    } else if (STEPS[S.step].scene === 'spiro') {
      const V = spV(SPR.u), f = dt > 0 ? (V - (SPR.pV ?? V)) / dt : 0; SPR.pV = V; SPR.dv = (SPR.dv || 0) + (clamp(f * 0.5, -2, 2) - (SPR.dv || 0)) * Math.min(1, dt * 6);
      S.airPh += SPR.dv * dt * 0.55;
    }
  }
  // где ручка (экран) и куда её тянуть — для попадания и для проверки
  function handle() {
    const I = S.ia; if (!I || S.fly || S.tr) return null;
    const k = S.step;
    if (I.kind === 'air') { const p = atLen(AP, APL, I.s * APT), q = AP[AP.length - 1]; return { p: toS(p[0], p[1]), to: toS(q[0], q[1]), r: 30 }; }
    if (I.kind === 'dia') { const y = diaY(-74, S.m.d, 0); return { p: toS(-74, y + 4), to: toS(-74, diaY(-74, 1, 0) + 4), r: 34 }; }
    if (I.kind === 'rib') { return { p: toS(0, -40 - S.m.r * 9), to: toS(0, -40 - 16), r: 32 }; }
    if (I.kind === 'tip') { const t = TREE.target; return { p: toS(t.x, t.y), to: toS(t.x, t.y), r: 30 }; }
    if (I.kind === 'o2' || I.kind === 'co2') { const M = G.mols[I.mol], R = RBC4(); return { p: toS(M.x, M.y), to: I.kind === 'o2' ? toS(R.x, R.y) : toS(-10, -40), r: 32 }; }
    if (I.kind === 'up' || I.kind === 'down') { const V = spV(SPR.u); return { p: toS(spX(SPR.u), spY(V)), to: toS(spX(I.kind === 'up' ? 3.8 : 5), spY(I.kind === 'up' ? SP.MAX : SP.MIN)), r: 32 }; }
    void k; return null;
  }
  function iaDown(p) {
    const I = S.ia, H = handle(); if (!I || !H || I.st === 'done' || I.st === 'auto') return false;
    if (hyp(p.x - H.p.x, p.y - H.p.y) > H.r) return false;
    if (I.kind === 'tip') { next(); return true; }
    I.st = 'drag'; I.p0 = p; I.v0 = I.v; S.labKey = null;
    if (I.kind === 'o2' || I.kind === 'co2') { const M = G.mols[I.mol], w = toW(p.x, p.y); M.st = 'drag'; I.gx = M.x - w.x; I.gy = M.y - w.y; }
    if (I.kind === 'up' || I.kind === 'down') I.u0 = SPR.u;
    return true;
  }
  function iaMove(p) {
    const I = S.ia; if (!I || I.st !== 'drag') return;
