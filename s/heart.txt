(() => {
  'use strict';

  const REDUCED = matchMedia('(prefers-reduced-motion: reduce)').matches;

  /* ---------- математика ---------- */
  const TAU = Math.PI * 2;
  const clamp = (v, a, b) => v < a ? a : v > b ? b : v;
  const clamp01 = v => v < 0 ? 0 : v > 1 ? 1 : v;
  const lerp = (a, b, k) => a + (b - a) * k;
  const smooth = v => { v = clamp01(v); return v * v * (3 - 2 * v); };
  const seg = (t, a, b) => smooth((t - a) / (b - a));
  const bump = (t, a, b, c, d) => seg(t, a, b) * (1 - seg(t, c, d));
  const hyp = Math.hypot;
  const FONT = (s, w = 700) => `${w} ${s}px JUZ, Inter, ui-sans-serif, system-ui, -apple-system, "Segoe UI", sans-serif`;
  const hash = (i, k = 0) => { const x = Math.sin(i * 127.1 + k * 311.7) * 43758.5453; return x - Math.floor(x); };
  const HEXC = {};
  function rgb(h) { let v = HEXC[h]; if (v) return v; if (h[0] !== '#') { v = HEXC[h] = h.match(/[\d.]+/g).slice(0, 3).map(Number); return v; } const n = parseInt(h.slice(1), 16); v = HEXC[h] = [n >> 16 & 255, n >> 8 & 255, n & 255]; return v; }
  function rgba(h, a) { const c = rgb(h); return `rgba(${c[0]},${c[1]},${c[2]},${a})`; }
  function mixC(a, b, k) { const A = rgb(a), B = rgb(b); k = clamp01(k); return `rgb(${Math.round(lerp(A[0], B[0], k))},${Math.round(lerp(A[1], B[1], k))},${Math.round(lerp(A[2], B[2], k))})`; }

  /* ---------- сызықтар: Катмулл — Ром қисықтары, түтіктер ---------- */
  function crTo(p, P, closed) {   // P арқылы өтетін тегіс қисық (Path2D не ctx)
    const n = P.length; if (n < 2) return;
    const g = i => closed ? P[(i + n) % n] : P[clamp(i, 0, n - 1)];
    p.moveTo(P[0][0], P[0][1]);
    const m = closed ? n : n - 1;
    for (let i = 0; i < m; i++) {
      const p0 = g(i - 1), p1 = g(i), p2 = g(i + 1), p3 = g(i + 2);
      p.bezierCurveTo(p1[0] + (p2[0] - p0[0]) / 6, p1[1] + (p2[1] - p0[1]) / 6, p2[0] - (p3[0] - p1[0]) / 6, p2[1] - (p3[1] - p1[1]) / 6, p2[0], p2[1]);
    }
    if (closed) p.closePath();
  }
  function shape(P) { const p = new Path2D(); crTo(p, P, true); return p; }
  function curve(P) { const p = new Path2D(); crTo(p, P, false); return p; }
  function crSample(P, per = 10, closed = false) {   // қисықтың нүктелері (бір рет, іске қосқанда)
    const n = P.length, g = i => closed ? P[(i + n) % n] : P[clamp(i, 0, n - 1)], out = [];
    const m = closed ? n : n - 1;
    for (let i = 0; i < m; i++) {
      const p0 = g(i - 1), p1 = g(i), p2 = g(i + 1), p3 = g(i + 2);
      for (let k = 0; k < per; k++) {
        const t = k / per, t2 = t * t, t3 = t2 * t;
        const f = (a, b, c, d) => 0.5 * ((2 * b) + (-a + c) * t + (2 * a - 5 * b + 4 * c - d) * t2 + (-a + 3 * b - 3 * c + d) * t3);
        const q = [f(p0[0], p1[0], p2[0], p3[0]), f(p0[1], p1[1], p2[1], p3[1])];
        for (let j = 2; j < p1.length; j++) q.push(lerp(p1[j], p2[j], t));
        out.push(q);
      }
    }
    if (!closed) out.push(P[n - 1].slice());
    return out;
  }
  function arcLen(S) { const L = [0]; for (let i = 1; i < S.length; i++) L.push(L[i - 1] + hyp(S[i][0] - S[i - 1][0], S[i][1] - S[i - 1][1])); return L; }
  function along(S, L, s) {   // доғаның s нүктесі: [x, y, бірлік жанама x, y, қосымша мәндер…]
    s = clamp(s, 0, L[L.length - 1]);
    let lo = 0, hi = L.length - 1; while (hi - lo > 1) { const m = (lo + hi) >> 1; if (L[m] <= s) lo = m; else hi = m; }
    const a = S[lo], b = S[hi], d = L[hi] - L[lo] || 1, k = (s - L[lo]) / d, dx = (b[0] - a[0]) / d, dy = (b[1] - a[1]) / d;
    return [lerp(a[0], b[0], k), lerp(a[1], b[1], k), dx, dy, a.length > 2 ? lerp(a[2], b[2], k) : 0];
  }
  // түтік: орта сызығы P, радиусы r — тамыр; бір рет құрылады
  function tube(P, r, per = 10) {
    const S = crSample(P, per), n = S.length, Lf = [], Rt = [];
    for (let i = 0; i < n; i++) {
      const a = S[Math.max(0, i - 1)], b = S[Math.min(n - 1, i + 1)], dx = b[0] - a[0], dy = b[1] - a[1], l = hyp(dx, dy) || 1;
      const nx = -dy / l, ny = dx / l, rr = typeof r === 'function' ? r(i / (n - 1)) : r;
      Lf.push([S[i][0] + nx * rr, S[i][1] + ny * rr]); Rt.push([S[i][0] - nx * rr, S[i][1] - ny * rr]);
    }
    const p = new Path2D(); p.moveTo(Lf[0][0], Lf[0][1]);
    for (let i = 1; i < n; i++) p.lineTo(Lf[i][0], Lf[i][1]);
    for (let i = n - 1; i >= 0; i--) p.lineTo(Rt[i][0], Rt[i][1]);
    p.closePath();
    const end = i => { const j = i === 0 ? 1 : n - 2, dx = S[i][0] - S[j][0], dy = S[i][1] - S[j][1], l = hyp(dx, dy) || 1; return { x: S[i][0], y: S[i][1], ux: dx / l, uy: dy / l, r: typeof r === 'function' ? r(i ? 1 : 0) : r }; };
    const hi = new Path2D(); for (let i = 0; i < n; i++) { const q = [lerp(S[i][0], Lf[i][0], 0.45), lerp(S[i][1], Lf[i][1], 0.45)]; if (i) hi.lineTo(q[0], q[1]); else hi.moveTo(q[0], q[1]); }
    return { p, S, e0: end(0), e1: end(n - 1), hi, r };
  }

  function lumenHalf(x, pl, th) {   // саңылаудың жарты биіктігі (жоғарғы шекарасы −rin; астыңғы — шытыра)

  function stepArtery(dt, pl, th) {

  function camAt(f, e) {
    const A = f.A, B = f.B, la = Math.log(A.z), lb = Math.log(B.z), z = Math.exp(lerp(la, lb, e));
    const w = Math.abs(lb - la) < 1e-3 ? e : (1 / A.z - 1 / z) / (1 / A.z - 1 / B.z);
    return { x: lerp(A.x, B.x, w), y: lerp(A.y, B.y, w), z };
  }

    const st = STEPS[k], rest = !!opts.rest, from = opts.from;
    S.actT = null; S.cutDone = false;
    if (st.lv === 1 && (!st.flow || opts.fade)) initParticles();

    const from = S.step, st = STEPS[k];
    if (S.paused) setPaused(false);
    const tgt = stepCam(k);
    if (opts.fade) { snapshot(); S.fly = null; CAM = tgt; S.arrT = S.t; }

      const dz = Math.abs(Math.log(tgt.z / CAM.z)), dist = hyp(tgt.x - CAM.x, tgt.y - CAM.y) * CAM.z;
      if (dz < 0.01 && dist < 2 && st.lv === STEPS[from].lv) { S.fly = null; CAM = tgt; S.arrT = S.t; }
      else S.fly = { A: { ...CAM }, B: tgt, t0: S.t, dur: clamp(0.9 + 0.38 * dz + dist / 900, 1.1, 3.4), la: STEPS[from].lv, lb: st.lv, ka: from, kb: k };

    S.step = k; S.labels = []; S.labKey = null;
