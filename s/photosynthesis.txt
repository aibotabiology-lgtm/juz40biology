  function stepPad(st, v) {
    const pad = { l: 12, r: 12, t: 12, b: 12 };
    if (st.tag) pad.t += 32;
    if (st.plate) pad.t = Math.max(pad.t, plateLayout(st.plate).h + 24);
    if (st === STEPS[13]) pad.b += narrow() ? 70 : 8;
    if (v.pad) for (const k in v.pad) pad[k] += v.pad[k];
    return pad;
  }
  function fitBox(b, pad) {
    const aw = Math.max(40, S.W - pad.l - pad.r), ah = Math.max(40, S.H - pad.t - pad.b);
    const z = Math.min(aw / (b.x1 - b.x0), ah / (b.y1 - b.y0));

  function advance(dt) {
    S.amb += dt; const v = S.paused ? 0 : dt * S.speed; S.t += v; if (!S.paused) S.clk += dt * S.speed;
    if (S.fly && S.t >= S.fly.t0 + S.fly.dur) { const K = S.fly.K; S.camL = { ...K[K.length - 1] }; S.fly = null; S.arrT = S.t; S.labKey = null; }
    if (S.prev && S.t >= S.prev.t0 + S.prev.dur) S.prev = null;
    if (!S.paused) actStep();
    if (S.step === 13 && !S.paused) {   // жарық — қор — Кальвин
      const L = S.light; L.J += ((L.on ? 1 : 0) - L.J) * Math.min(1, v * 6);
      L.C = clamp01(L.A / 0.3); L.A = clamp(L.A + (0.3 * L.J * (1 - L.A) - 0.12 * L.C) * v, 0, 1); L.ph = (L.ph || 0) + L.C * v * 1.4;
    }

    if (calm) {

  }

  }
