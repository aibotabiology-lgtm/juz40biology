  function gridIn(R, n, i, colW = 82, rowH = 54) {
    const w = R.x1 - R.x0, h = R.y1 - R.y0, cols = Math.max(1, Math.min(n, Math.floor(w / colW))), rows = Math.ceil(n / cols);
    const r = Math.floor(i / cols), c = i % cols, inRow = r === rows - 1 ? n - r * cols : cols;
    const gy = Math.min(rowH, h / rows);
    return { x: R.x0 + w * (c + 0.5) / inRow, y: R.y0 + (h - gy * rows) / 2 + gy * (r + 0.4) };
  }
  function homeOf(T, i) {
    const D = S.drag; if (!D) return { x: 0, y: 0 };

    return gridIn(wrect(D.geo.src), D.toks.length, i);

  function setupDrag(k) {
    const st = STEPS[k];
    if (!st.drag) { S.drag = null; return; }
    let slot = 0;
    S.drag = { kind: st.drag, geo: dragGeo(st.drag), toks: TOKS[st.drag].map((d, i) => ({ ...d, i, st: 'home', slot: d.ok ? slot++ : -1, x: 0, y: 0, t0: 0 })), cur: null, doneT: null, tries: 0 };
    if (st.drag === 'adh') S.adh = { k: 0, on: false };
  }
  function tokPos(T) {
    const D = S.drag;
    if (T.st === 'drag') return { x: T.x, y: T.y };
    if (T.st === 'home') { const h = homeOf(T, T.i), b = Math.sin(S.clk * 2 + T.i * 1.7) * 2.2; return { x: h.x, y: h.y + b }; }
    if (T.st === 'back') { const h = homeOf(T, T.i), q = ease2((S.t - T.t0) / 0.5); return { x: lerp(T.x, h.x, q), y: lerp(T.y, h.y, q) }; }
    const s = slotOf(T);
    if (T.st === 'fly') {
      const q = ease2((S.t - T.t0) / T.dur);
      const via = D.kind === 'adh' ? { x: s.x, y: s.y } : { x: lerp(T.x, s.x, 0.5), y: toS(S.cam, 0, D.geo.viaY).y };
      if (q < 0.5) { const u = q / 0.5; return { x: lerp(T.x, via.x, u), y: lerp(T.y, via.y, u) }; }
      const u = (q - 0.5) / 0.5; return { x: lerp(via.x, s.x, u), y: lerp(via.y, s.y, u) };
    }
    const b = Math.sin(S.clk * 1.6 + T.i) * 1.5; return { x: s.x, y: s.y + b };
  }
  function tokHit(p) {
    const D = S.drag; if (!D || S.fly || S.trans || S.auto) return null;
    let best = null, bd = 1e9;
    for (const T of D.toks) { if (T.st !== 'home') continue; const q = tokPos(T), d = hyp(q.x - p.x, q.y - p.y), lim = (T.big ? 26 : 22) + 6; if (d < lim && d < bd) { bd = d; best = T; } }
    return best;
  }
  function inDst(p) {
    const D = S.drag, w = toW(S.cam, p.x, p.y);
    if (D.kind === 'adh') return Math.abs(w.x - 206) < 40 && w.y > -200 && w.y < 330;
    if (D.kind === 'filter') return w.y > FILT.Y3;
    if (D.kind === 'reabs') return w.y > TUB.YC - 10;
    return w.y < TUB.YL + 6;
  }
  function launch(T, from, dur) { T.st = 'fly'; T.x = from.x; T.y = from.y; T.t0 = S.t; T.dur = dur || (S.drag.kind === 'adh' ? 0.6 : 1.1); }
  function dropTok(T) {
    const D = S.drag, p = { x: T.x, y: T.y };

      else { T.st = 'back'; T.t0 = S.t; toast(T.msg); }
    } else { T.st = 'back'; T.t0 = S.t; if (D.kind === 'filter' && T.big) toast(T.msg); }
  }

  function stepDrag() {
    const D = S.drag; if (!D) return;
    for (const T of D.toks) {
      if (T.st === 'fly' && S.t - T.t0 >= T.dur) T.st = 'done';
      if (T.st === 'back' && S.t - T.t0 >= 0.5) T.st = 'home';
    }
    if (S.adh && S.adh.on) S.adh.k = clamp01((S.t - S.adh.t0) / 2.2);

  };

  };
